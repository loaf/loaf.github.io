1.搭应用框架

因为我选择的技术栈是Tauri+React 前端+FastAPI后端，所以我们先得将这个框架跑通。

这里要注意的是，React前端，与上面说的智能体中ReAct是完全不同的两回事。前者是由Meta开发的用户界面JavaScript库，后者是AI开发的工程范式，不是一个代码库，而是一种编码理念，体现在FastAPI后端（或者单独调用的LLM服务中）。

//! MewGo Tauri 壳 —— 入口。
//!
//! 1. spawn Python sidecar (uv → uvicorn)
//! 2. 轮询 /health 等它就绪
//! 3. 注册 IPC, 开窗口
//! 4. 退出时 kill sidecar

mod commands;
mod sidecar;

use sidecar::SidecarGuard;
use std::sync::Arc;
use tauri::Manager;

#[cfg_attr(mobile, tauri::mobile_entry_point)]
pub fn run() {
    tauri::Builder::default()
        .manage(SidecarGuard::empty())
        .setup(|app| {
            // backend 目录 = 应用资源目录(开发期是 src-tauri)的 ../backend
            // 开发期: 当前工作目录是 src-tauri,所以 ../backend 即可。
            let backend_dir = std::env::current_dir()
                .map(|d| d.join("../backend"))
                .map(|p| p.to_string_lossy().to_string())
                .unwrap_or_else(|_| "../backend".to_string());

            match sidecar::spawn_sidecar(&backend_dir) {
                Ok(child) => {
                    let guard = app.state::<SidecarGuard>();
                    guard.set_child(child);
                    println!("[mewgo] sidecar spawned, waiting for /health...");
                    match sidecar::wait_until_healthy(60) {
                        Ok(()) => println!("[mewgo] backend healthy ✓"),
                        Err(e) => {
                            eprintln!("[mewgo] WARN backend not healthy: {e}");
                            eprintln!("[mewgo] 应用仍会启动,但后端调用可能失败。");
                        }
                    }
                }
                Err(e) => {
                    eprintln!("[mewgo] 启动 sidecar 失败: {e}");
                    eprintln!("[mewgo] 请确认 uv 在 PATH 中,且 backend/pyproject.toml 已 `uv sync`。");
                }
            }
            Ok(())
        })
        .invoke_handler(tauri::generate_handler![commands::get_backend_base_url])
        .on_window_event(|window, event| {
            if let tauri::WindowEvent::CloseRequested { .. } = event {
                if let Some(guard) = window.app_handle().try_state::<SidecarGuard>() {
                    guard.kill();
                    println!("[mewgo] sidecar killed on window close");
                }
            }
        })
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}

// 让 Arc<SidecarGuard> 不触发未使用警告(留给后续卡用)
#[allow(dead_code)]
type _SharedGuard = Arc<SidecarGuard>;

3-5）src-tauri/src/sidecar.rs ， 后端进程管理：

//! Sidecar 生命周期管理。
//!
//! 流程:
//! 1. spawn `uv run --project <backend> uvicorn app.main:app --port 8000`
//! 2. 轮询 `GET /health` 直到通(或超时)
//! 3. 把 child guard 挂到 state; 退出时 kill

use std::process::{Child, Command, Stdio};
use std::sync::Mutex;

/// 后端 sidecar 监听端口。
pub const BACKEND_PORT: u16 = 8000;
pub const BACKEND_BASE_URL: &str = "http://localhost:8000";

/// 持有 sidecar 子进程的 guard。显式 kill 时终结子进程。
pub struct SidecarGuard {
    child: Mutex<Option<Child>>,
}

impl SidecarGuard {
    /// 占位构造,供 tauri::State 初始化用; 实际 child 在 setup 里替换。
    pub fn empty() -> Self {
        Self {
            child: Mutex::new(None),
        }
    }

    /// 把 spawn 出的 child 注入(setup 阶段用)。
    pub fn set_child(&self, child: Child) {
        if let Ok(mut guard) = self.child.lock() {
            *guard = Some(child);
        }
    }

    pub fn kill(&self) {
        if let Ok(mut guard) = self.child.lock() {
            if let Some(mut child) = guard.take() {
                let _ = child.kill();
                let _ = child.wait();
            }
        }
    }
}

/// spawn Python sidecar。返回 child。
///
/// backend 目录相对 src-tauri 是 `../backend`。
/// `uv run --project <dir>` 会用该目录的 pyproject + venv。
pub fn spawn_sidecar(backend_dir: &str) -> std::io::Result<Child> {
    let port = BACKEND_PORT.to_string();
    let mut cmd = Command::new("uv");
    cmd.args([
        "run",
        "--project",
        backend_dir,
        "uvicorn",
        "app.main:app",
        "--port",
        &port,
    ])
    .stdin(Stdio::null())
    .stdout(Stdio::inherit())
    .stderr(Stdio::inherit());

    // Windows 上隐藏控制台窗口
    #[cfg(windows)]
    {
        use std::os::windows::process::CommandExt;
        // CREATE_NO_WINDOW = 0x08000000
        cmd.creation_flags(0x08000000);
    }

    cmd.spawn()
}

/// 轮询 `/health` 直到通或超时(秒)。
pub fn wait_until_healthy(timeout_secs: u64) -> Result<(), String> {
    let url = format!("{}/health", BACKEND_BASE_URL);
    let start = std::time::Instant::now();
    let timeout = std::time::Duration::from_secs(timeout_secs);
    loop {
        if start.elapsed() > timeout {
            return Err(format!(
                "backend did not become healthy within {}s",
                timeout_secs
            ));
        }
        match reqwest::blocking::get(&url) {
            Ok(resp) if resp.status().is_success() => return Ok(()),
            _ => std::thread::sleep(std::time::Duration::from_millis(500)),
        }
    }
}

2. 使用LiteLLM加上AI对话功能

在上面的工作完成后，应用能开窗口、能调 /health。但后端只会回 “ok”。现在先加上LLM试一下。

调用 LLM 用 LiteLLM，这是一个可统一多家大模型供应商的Python库。

1. 处理后端

后端代码的结构如下：

backend/app/
├── config.py            # 读 .env / 环境变量, 拿到 API key
├── llm/
│   └── router.py        # LiteLLM 封装: chat() -> 流式 yield token
└── api/
    └── chat.py          # POST /chat, 把 LLM 流包成 SSE 返回
config.py，配置层
from __future__ import annotations

from functools import lru_cache

from pydantic_settings import BaseSettings, SettingsConfigDict


class Settings(BaseSettings):
    """应用配置。
    优先级: 环境变量 > backend/.env 文件 > 默认值。
    """

    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        extra="ignore",
    )

    # DeepSeek (OpenAI 兼容)。LiteLLM 用 deepseek/ 前缀识别。
    deepseek_api_key: str = ""
    deepseek_api_base: str = "https://api.deepseek.com/v1"
    deepseek_model: str = "deepseek-chat"

    # OpenAI 协议兼容的通用回退(若用户用别的兼容服务)。
    # LiteLLM 读 OPENAI_API_KEY / OPENAI_API_BASE / OPENAI_BASE_URL 也行,
    # 这里显式列出方便排查。
    openai_api_key: str = ""

    @property
    def has_llm_key(self) -> bool:
        """是否配了至少一个 LLM key。"""
        return bool(self.deepseek_api_key or self.openai_api_key)


@lru_cache
def get_settings() -> Settings:
    """单例: 全进程只读一次 .env。"""
    return Settings()
llm/router.py ， LLM 调用核心：
"""LLM 路由器 —— 流式对话。

用 LiteLLM 调 DeepSeek (OpenAI 兼容协议), 流式返回 token。
DeepSeek 在 LiteLLM 里用 `deepseek/` 前缀识别, 会自动走 OpenAI 协议。

"""

from __future__ import annotations

from collections.abc import AsyncIterator

import litellm

from app.config import get_settings


def _build_completion_kwargs(
    messages: list[dict[str, str]],
    model: str | None,
) -> dict:
    """组装 LiteLLM acompletion 的参数。

    把 key/base_url/model 从配置注入, 调用方只关心 messages。
    """
    settings = get_settings()
    chosen_model = model or settings.deepseek_model
    kwargs: dict = {
        "model": f"deepseek/{chosen_model}",
        "messages": messages,
        "stream": True,
    }
    if settings.deepseek_api_key:
        kwargs["api_key"] = settings.deepseek_api_key
    if settings.deepseek_api_base:
        kwargs["api_base"] = settings.deepseek_api_base
    return kwargs


async def chat(
    messages: list[dict[str, str]],
    model: str | None = None,
) -> AsyncIterator[str]:
    """流式对话, 逐个 token chunk yield 文本。

    参数:
        messages: OpenAI 消息格式 [{"role":"user","content":"..."}]
        model: 可选模型名覆盖(默认走配置里的 deepseek_model)

    yield:
        str —— 每次吐一小段文本(可能是一个字、一个词、一句话)

    异常:
        RuntimeError —— 没配 API key 时
        litellm 的各种异常 —— 调用失败时(后续 T-020 加重试)
    """
    settings = get_settings()
    if not settings.has_llm_key:
        raise RuntimeError(
            "未配置 LLM API Key。请在 backend/.env 设置 DEEPSEEK_API_KEY "
            "(参考 .env.example)。"
        )

    kwargs = _build_completion_kwargs(messages, model)
    # litellm.acompletion(stream=True) 返回一个 async generator,
    # 每个 chunk 是 OpenAI 格式的 ChatCompletionChunk。
    response = await litellm.acompletion(**kwargs)
    async for chunk in response:
        # chunk.choices[0].delta.content 可能是 None(首个 chunk 只有 role)
        try:
            delta = chunk.choices[0].delta.content
        except (IndexError, AttributeError):
            delta = None
        if delta:
            yield delta

这里的AsyncIterator + yield 在Python中，不是直接return一个值，而是一边算一边吐出多个值。调用方可以用async for token in chat(...) 一个一个接收。这正是流式的本质。

api/chat.py,SSE接口
"""聊天接口 —— POST /chat, SSE 流式返回。

SSE (Server-Sent Events) 格式: 每条消息是 `data: <payload>\\n\\n`。
前端用 fetch + ReadableStream 读(EventSource 只支持 GET, POST 用不了)。
"""

from __future__ import annotations

import json
from collections.abc import AsyncIterator

from fastapi import APIRouter, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel

from app.llm import chat as llm_chat

router = APIRouter()


class ChatRequest(BaseModel):
    """单轮对话请求。"""

    message: str


def _sse(data: dict | str) -> str:
    """把 payload 包成一条 SSE 数据行。"""
    payload = data if isinstance(data, str) else json.dumps(data, ensure_ascii=False)
    return f"data: {payload}\n\n"


async def _stream(messages: list[dict[str, str]]) -> AsyncIterator[str]:
    """生成 SSE 流: 先发 role 标记, 逐 chunk 吐文本, 末尾 [DONE]。"""
    try:
        async for token in llm_chat(messages):
            yield _sse({"chunk": token})
    except RuntimeError as e:
        # 配置问题(没 key 等), 以 error 事件返回, 不让连接挂死
        yield _sse({"error": str(e)})
    except Exception as e:  # noqa: BLE001        
        yield _sse({"error": f"LLM 调用失败: {e}"})
    yield _sse("[DONE]")


@router.post("/chat")
async def chat(req: ChatRequest) -> StreamingResponse:
    """单轮流式对话。

    把用户消息包成 OpenAI messages 格式喂给 router。
    """
    if not req.message.strip():
        raise HTTPException(status_code=400, detail="message 不能为空")

    messages = [
        {"role": "system", "content": "你是 MewGo 的写作助手。回答简洁。"},
        {"role": "user", "content": req.message},
    ]
    return StreamingResponse(
        _stream(messages),
        media_type="text/event-stream",
        headers={
            # 禁用 nginx 等中间件缓冲, 保证流式实时性
            "Cache-Control": "no-cache",
            "X-Accel-Buffering": "no",
        },
    )

FastAPI 的 StreamingResponse 接受一个生成器，会把生成器 yield 的每个字符串立即推给客户端（不缓存）。这是"流式 HTTP"在 Python 侧的实现方式。

import { getHealth, streamChat } from "./api/client";

// 健康按钮 + 一个流式聊天框。
// 聊天框验证"前端 → 后端 → LLM 流式返回"链路。
type Msg = { role: "user" | "assistant"; text: string };

export default function App() {
  const [status, setStatus] = useState<string | null>(null);
  const [healthErr, setHealthErr] = useState<string | null>(null);

  const [input, setInput] = useState("");
  const [messages, setMessages] = useState<Msg[]>([]);
  const [streaming, setStreaming] = useState(false);
  const [chatErr, setChatErr] = useState<string | null>(null);

  async function checkHealth() {
    setHealthErr(null);
    setStatus(null);
    try {
      const data = await getHealth();
      setStatus(data.status);
    } catch (e) {
      setHealthErr(e instanceof Error ? e.message : String(e));
    }
  }

  async function send() {
    const text = input.trim();
    if (!text || streaming) return;
    setInput("");
    setChatErr(null);
    // 先把用户消息和空的 assistant 占位塞进去, 再流式填充后者
    setMessages((m) => [...m, { role: "user", text }, { role: "assistant", text: "" }]);
    setStreaming(true);

        // 流式 append 到最后一条(assistant 占位)

          const next = [...m];
          const last = next[next.length - 1];
          next[next.length - 1] = { ...last, text: last.text + chunk };
          return next;
        });
      });
    } catch (e) {
      setChatErr(e instanceof Error ? e.message : String(e));
    } finally {
      setStreaming(false);
    }
  }

      </section>

      {/* 流式聊天 */}
      <section>
        <h2>聊天 (流式)</h2>

          {messages.length === 0 && <p style={{ color: "#999" }}>发条消息试试…</p>}

              <strong>{m.role === "user" ? "我" : "AI"}:</strong> {m.text}
              {streaming && i === messages.length - 1 && m.role === "assistant" && (
                <span style={{ color: "#999" }}>▌</span>

// 后端 API 客户端。
// 基地址优先从 Tauri IPC 拿(Rust 侧 `get_backend_base_url` 命令);
// 若不在 Tauri 环境(纯浏览器 `vite` 调试),降级到 localhost:8000。


const FALLBACK_BASE_URL = "http://localhost:8000";

let cachedBaseUrl: string | null = null;

async function resolveBaseUrl(): Promise<string> {
  if (cachedBaseUrl) return cachedBaseUrl;

  // 动态 import: 纯浏览器调试时 @tauri-apps/api 也装着,但 invoke 会失败,
  // 用 try/catch 降级即可。
  try {
    const { invoke } = await import("@tauri-apps/api/core");
    const url = await invoke<string>("get_backend_base_url");
    cachedBaseUrl = url;
    return url;
  } catch {
    cachedBaseUrl = FALLBACK_BASE_URL;
    return cachedBaseUrl;
  }
}

export async function getHealth(): Promise<{ status: string }> {
  const base = await resolveBaseUrl();
  const res = await fetch(`${base}/health`);
  if (!res.ok) {
    throw new Error(`health check failed: HTTP ${res.status}`);
  }
  return res.json();
}

/**
 * 流式聊天 —— POST /chat, 读 SSE 流。
 *
 * 为什么用 fetch + ReadableStream 而不是 EventSource?
 *   EventSource 只支持 GET 请求, 而 /chat 是 POST(要带 body)。
 *   所以手动用 fetch 拿到 response.body 这个 ReadableStream, 逐块解析 SSE。
 *
 * @param message 用户输入
 * @param onChunk 每收到一段文本就回调(用来实时 append 到 UI)
 */
export async function streamChat(
  message: string,
  onChunk: (text: string) => void
): Promise<void> {
  const base = await resolveBaseUrl();
  const res = await fetch(`${base}/chat`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ message }),
  });

  if (!res.ok || !res.body) {
    throw new Error(`chat failed: HTTP ${res.status}`);
  }

  const reader = res.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";

  // SSE 以 \n\n 分隔每条事件, 一条事件形如 `data: {...}\n\n`。
  // 但网络分块不保证按事件边界切, 所以要先攒到 buffer 再按 \n\n 切。
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });

    const parts = buffer.split("\n\n");
    // 最后一段可能不完整, 留给下一轮
    buffer = parts.pop() ?? "";

    for (const part of parts) {
      const line = part.trim();
      if (!line.startsWith("data:")) continue;
      const payload = line.slice(5).trim();
      if (payload === "[DONE]") return;
      try {
        const data = JSON.parse(payload);
        if (data.chunk) onChunk(data.chunk);
        if (data.error) throw new Error(data.error);
      } catch (e) {
        // JSON 解析失败就跳过这条; 但 error 类要抛
        if (e instanceof Error && e.message.includes("调用"))
          throw e;
      }
    }
  }
}

这里使用 buffer 是因为网络数据包不能保证SSE事件边界，一条完整 SSE 事件是 data: {...}\n\n，但网络可能把半条事件分两次送来。 所以先攒到 buffer，按 \n\n 切，最后一段（可能不完整）留到下次。
